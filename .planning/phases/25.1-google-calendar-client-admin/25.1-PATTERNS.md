export async function exchangeAuthCodeForTokens(
  code: string
): Promise<{ refreshToken: string; accessToken: string }> {
  const { tokens } = await getOAuth2Client().getToken(code);
  if (!tokens.refresh_token) {
    throw new Error(
      "Google did not return a refresh token -- ensure prompt=consent was used and this is the account's first authorization"
    );
  }
  return { refreshToken: tokens.refresh_token, accessToken: tokens.access_token ?? '' };
}

import crypto from 'crypto';

export function generateOAuthState(): string {
  // 16 bytes (128 bits) of random data, hex-encoded = 32 hex chars
  return crypto.randomBytes(16).toString('hex');
}

export async function validateOAuthState(state: string, stored: string): boolean {
  // Constant-time comparison to prevent timing attacks
  return crypto.timingSafeEqual(Buffer.from(state), Buffer.from(stored));
}

```typescript
import { Request, Response } from 'express';
import { findBusinessByOwnerTelegramId } from '../onboarding/queries';
import { sendTelegramMessage } from '../telegram/client';

// In-memory state store (PoC only; production uses Redis/DB with TTL)
const oauthStateStore = new Map<string, { businessId: number; createdAt: Date }>();

export async function handleGoogleOAuthCallback(req: Request, res: Response): Promise<void> {
  const state = req.query.state as string | undefined;
  const code = req.query.code as string | undefined;
  const error = req.query.error as string | undefined;

  // CSRF validation
  if (!state || !oauthStateStore.has(state)) {
    logger.warn({ state }, 'OAuth callback: invalid or missing state');
    res.status(400).send('Invalid state — possible CSRF attack.');
    return;
  }

  oauthStateStore.delete(state); // One-time use only

  // Handle error response
  if (error) {
    logger.warn({ error }, 'Google OAuth consent denied by user');
    res.status(400).send(`Authorization declined: ${error}.`);
    return;
  }

  // Code validation
  if (!code) {
    res.status(400).send('Missing authorization code.');
    return;
  }

  try {
    const { refreshToken } = await exchangeAuthCodeForTokens(code);
    const ownerBusiness = await findBusinessByOwnerTelegramId(ownerTelegramId);
    if (!ownerBusiness) {
      res.status(403).send('Business not found.');
      return;
    }
    await storeGoogleRefreshToken(ownerBusiness.id, refreshToken);
    await sendTelegramMessage(ownerTelegramId, '✅ Google Calendar συνδέθηκε με επιτυχία!');
    res.status(200).send('<html>...</html>');
  } catch (err) {
    logger.error({ err, state }, 'OAuth callback: token exchange failed');
    res.status(500).send('Setup failed — see server logs.');
  }
}
```
